// Proc33: vyznachennia koordynatnykh chvertei
int Quarter(double x, double y)
{
    if (x > 0 && y > 0) return 1;
    if (x < 0 && y > 0) return 2;
    if (x < 0 && y < 0) return 3;
    return 4;
}

        !ReadDouble(x3, "x3: ") || !ReadDouble(y3, "y3: "))
        return;

// Proc23: obchyslennia koreniv kvadratnoho rivniannia
void QuadraticRoots(double a, double b, double c,
                    double& x1, double& x2, bool& hasRoots)
{
    double D = b * b - 4 * a * c;

    hasRoots = (D >= 0);

    if (hasRoots)
    {
        x1 = (-b + sqrt(D)) / (2 * a);
        x2 = (-b - sqrt(D)) / (2 * a);
    }
}

// Dopomizhna funktsiia dlia vvedennia chysel
bool ReadDouble(double& x, const string& message)
{
    cout << message;

    if (!(cin >> x) || !isfinite(x) || x < -100 || x > 100)
    {
        cout << "Pomylka! Vvedit chyslo vid -100 do 100." << endl;
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        return false;
    }

    return true;
}

// Vvod koefitsiientiv i rozviazannia rivniannia
void SolveEquation(int number)
{
    double a, b, c, x1 = 0, x2 = 0;
    bool hasRoots;

    cout << "\nRivniannia " << number << ":" << endl;

    if (!ReadDouble(a, "a = ") ||
        !ReadDouble(b, "b = ") ||
        !ReadDouble(c, "c = "))
        return;

    if (a == 0)
    {
        cout << "Pomylka! Koefitsiient a ne mozhe dorivniuvaty nuliu." << endl;
        return;
    }

    QuadraticRoots(a, b, c, x1, x2, hasRoots);

    if (hasRoots)
        cout << "Koreni: x1 = " << x1 << ", x2 = " << x2 << endl;
    else
        cout << "Diisnykh koreniv nemaie." << endl;
}

bool InputYear(int& year)
{
    cout << "\nVvedit rik: ";

    if (!(cin >> year) || year <= 0)
    {
        cout << "Pomylka! Rik maie buty dodatnim tsilym chyslom." << endl;
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
        return false;
    }

    return true;
}

// Obchyslennia nomera stolittia
int CalculateCentury(int year)
{
    return (year - 1) / 100 + 1;
}

// Vyvedennia rezultatu
void OutputCentury(int year, int century)
{
    cout << "Rik " << year << " nalezhyt do "
         << century << " stolittia." << endl;
}

void task3()
{
    int year;

    cout << "\nInteger30. Vyznachennia stolittia." << endl;

    if (InputYear(year))
        OutputCentury(year, CalculateCentury(year));
}
